    35→export function ExplainabilityCenter() {
    36→  const globalQuery = useQuery({
    37→    queryKey: ["explain-global"],
    38→    queryFn: ({ signal }) => socApi.explainGlobal(signal),
    39→  });
    40→  const modelQuery = useQuery({
    41→    queryKey: ["model-info"],
    42→    queryFn: ({ signal }) => socApi.modelInfo(signal),
    43→  });
    44→
    45→  if (globalQuery.isPending || modelQuery.isPending) {
    46→    return (
    47→      <div className="space-y-6">
    48→        <div className="grid gap-4 lg:grid-cols-2">
    49→          <Skeleton className="h-96 rounded-xl" />
    50→          <Skeleton className="h-96 rounded-xl" />
    51→        </div>
    52→        <Skeleton className="h-96 rounded-xl" />
    53→      </div>
    54→    );
    55→  }
    56→
    57→  if (globalQuery.isError || !globalQuery.data) {
    58→    return (
    59→      <ErrorState
    60→        title="Explainability data unavailable"
    61→        description="Could not load global SHAP / calibration artifacts from the inference engine."
    62→        onRetry={() => void globalQuery.refetch()}

    64→      />
    65→    );

    87→function GlobalImportance({ data }: { data: GlobalData }) {

    91→        .sort((a, b) => b.mean_abs_shap - a.mean_abs_shap)

    97→    () =>
    98→      [...(data.multiclassGain?.features ?? [])]
    99→        .sort((a, b) => b.gain - a.gain)

   101→        .reverse(),

   403→function LocalExplorer() {
   404→  const [category, setCategory] = useState<string>("Exploits");
   405→  const [selectedEventId, setSelectedEventId] = useState<string | null>(null);
   406→
   407→  const eventsQuery = useQuery({
   408→    queryKey: ["events", category],
   409→    queryFn: ({ signal }) => socApi.events({ category, n: 25 }, signal),
   410→  });
   411→
   412→  const explainMutation = useMutation({
   413→    mutationFn: (raw: Parameters<typeof socApi.explainEvent>[0]) => socApi.explainEvent(raw),
   414→  });
   415→
   416→  const events = eventsQuery.data?.events ?? [];
   417→  const selected = events.find((e) => e.eventId === selectedEventId) ?? null;
   418→  const explained = explainMutation.data ?? null;
   419→
   420→  const runExplain = (eventId: string) => {
   421→    setSelectedEventId(eventId);
   422→    const ev = events.find((e) => e.eventId === eventId);
   423→    if (ev) explainMutation.mutate(ev.raw);
   424→  };
   425→
   426→  return (
   427→    <section aria-label="local explanation explorer">
   428→      <SectionHeader
   429→        title="Local Explanation Explorer"
   430→        hint="pick a real test-set event and run it through the live engine"
   431→        icon={<Microscope className="h-4 w-4" aria-hidden />}
   432→      />
   433→      <Card className="gap-0 rounded-xl border-border/50 bg-card/50 p-4 transition-colors hover:border-primary/30">
   434→        <div className="mb-3 flex flex-wrap items-center gap-2">
   435→          <Select value={category} onValueChange={(v) => { setCategory(v); setSelectedEventId(null); }}>
   436→            <SelectTrigger className="h-9 w-56 border-border/60 bg-background/50 text-xs" aria-label="Select attack category">
   437→              <SelectValue />
   438→            </SelectTrigger>
   439→            <SelectContent className="soc-scroll max-h-72">
   440→              {UNSW_CATEGORIES.map((c) => (
   441→                <SelectItem key={c} value={c} className="text-xs">
   442→                  {c}
   443→                </SelectItem>
   444→              ))}
   445→            </SelectContent>
   446→          </Select>
   447→          <span className="font-mono text-[10px] text-muted-foreground">
   448→            GET /api/events?category={category}&n=25 — deterministic real test rows, scored live
   449→          </span>
   450→        </div>
   451→
   452→        {eventsQuery.isPending && <Skeleton className="h-24 rounded-lg" />}
   453→        {eventsQuery.isError && (
   454→          <ErrorState
   455→            title="Could not load events"
   456→            description="The engine did not return test-set events for this category."
   457→            onRetry={() => void eventsQuery.refetch()}
   458→          />
   459→        )}
   460→
   461→        {eventsQuery.isSuccess && (
   462→          <>
   463→            <div className="soc-scroll mb-4 flex max-h-28 flex-wrap gap-1.5 overflow-y-auto pr-1">
   464→              {events.map((e) => (
   465→                <button
   466→                  key={e.eventId}
   467→                  type="button"
   468→                  onClick={() => runExplain(e.eventId)}
   469→                  aria-pressed={selectedEventId === e.eventId}
   470→                  className={`rounded-md border px-2 py-1 font-mono text-[10px] transition-colors ${
   471→                    selectedEventId === e.eventId
   472→                      ? "border-primary/50 bg-primary/15 text-primary"
   473→                      : "border-border/50 bg-background/40 text-muted-foreground hover:border-primary/30 hover:text-foreground/85"
   474→                  }`}
   475→                >
   476→                  {e.eventId}
   477→                  <span className="ml-1.5" style={{ color: e.severity === "Critical" ? "#f43f5e" : "#8b98a9" }}>
   478→                    {fmtRisk(e.riskScore)}
   479→                  </span>
   480→                </button>
   481→              ))}
   482→              {events.length === 0 && (
   483→                <p className="text-xs text-muted-foreground">No events found for this category.</p>
   484→              )}
   485→            </div>
   486→
   487→            {!selected && !explainMutation.isPending && (
   488→              <p className="py-6 text-center text-xs text-muted-foreground">
   489→                Select an event chip above to score it live via POST /api/explain and get a full local explanation.
   490→              </p>
   491→            )}
   492→
   493→            {(explainMutation.isPending || explained) && selected && (
   494→              <div>
   495→                {explainMutation.isPending && <Skeleton className="h-64 rounded-lg" />}
   496→                {explained && <LocalExplanationResult event={explained} />}
   497→              </div>
   498→            )}
   499→            {explainMutation.isError && selected && (
   500→              <ErrorState
   501→                title="Explanation failed"
   502→                description="POST /api/explain did not succeed. Retry."
   503→                onRetry={() => selected && explainMutation.mutate(selected.raw)}
   504→              />
   505→            )}
   506→          </>
   507→        )}
   508→      </Card>
   509→    </section>
   510→  );
   511→}

   513→function LocalExplanationResult({ event }: { event: ScoredEvent }) {
   514→  const expl = event.explanation;
   515→  const scale = maxContribution([...expl.topPositive, ...expl.topNegative]);
   516→  const probs = Object.entries(event.categoryProbs ?? {}).sort((a, b) => b[1] - a[1]);
   517→
   518→  return (
   519→    <div className="space-y-4">
   520→      <div className="grid gap-4 lg:grid-cols-[auto_1fr]">
   521→        <div className="flex flex-col items-center gap-2">
   522→          <RiskGauge value={event.riskScore} size={128} label="risk" />
   523→          <SeverityBadge severity={event.severity} />
   524→        </div>
   525→        <div className="space-y-3">
   526→          <div className="flex flex-wrap items-center gap-2">
   527→            <span className="font-mono text-sm font-semibold text-primary">{event.eventId}</span>

   689→interface ColumnDef {
   690→  key: string;
   691→  label: string;
   692→  format?: (v: number) => string;
   693→  highlight?: boolean;
   694→}
   695→
   696→function ComparisonTable({
   697→  rows,
   698→  columns,
   699→  bestKey,
   700→}: {
   701→  rows: Record<string, unknown>[];
   702→  columns: ColumnDef[];
   703→  bestKey: string;
   704→}) {
   705→  const bestVal = Math.max(...rows.map((r) => Number(r[bestKey] ?? 0)));
   706→  return (
   707→    <div className="soc-scroll overflow-x-auto">
   708→      <table className="w-full text-left text-xs">
   709→        <thead>
   710→          <tr className="border-b border-border/50 text-[10px] uppercase tracking-wider text-muted-foreground">
   711→            {columns.map((c) => (
   712→              <th key={c.key} className={`px-2 py-2 font-medium ${c.key === "model" ? "" : "text-right"}`}>
   713→                {c.label}
   714→              </th>
   715→            ))}
   716→          </tr>
   717→        </thead>
   718→        <tbody>
   719→          {rows.map((row, i) => {
   720→            const isBest = Number(row[bestKey] ?? 0) === bestVal;
   721→            return (
   722→              <tr
   723→                key={i}
   724→                className={`border-b border-border/25 ${isBest ? "bg-primary/5" : ""}`}
   725→              >
   726→                {columns.map((c) => {
   727→                  const v = row[c.key];
   728→                  if (c.key === "model") {
   729→                    return (
   730→                      <td key={c.key} className="px-2 py-1.5">
   731→                        <span className="inline-flex items-center gap-1.5 text-foreground/90">
   732→                          {String(v)}
   733→                          {isBest && (
   734→                            <Badge variant="outline" className="border-primary/30 bg-primary/10 text-[8px] uppercase tracking-wider text-primary">
   735→                              selected
   736→                            </Badge>
   737→                          )}
   738→                        </span>
   739→                      </td>
   740→                    );
   741→                  }
   742→                  const num = Number(v);
   743→                  return (
   744→                    <td
   745→                      key={c.key}
   746→                      className={`px-2 py-1.5 text-right font-mono tabular-nums ${
   747→                        c.highlight && isBest ? "font-bold text-primary" : "text-muted-foreground"
   748→                      }`}
   749→                    >
   750→                      {Number.isFinite(num) && v !== undefined && v !== null ? c.format?.(num) ?? num.toFixed(3) : "—"}
   751→                    </td>
   752→                  );
   753→                })}
   754→              </tr>
   755→            );
   756→          })}
   757→        </tbody>
   758→      </table>
   759→    </div>
   760→  );
   761→}
   762→
   763→function Row({ label, value, mono = false }: { label: string; value: string; mono?: boolean }) {
   764→  return (
   765→    <div className="flex items-center justify-between gap-3 border-b border-border/25 pb-1">
   766→      <dt className="text-muted-foreground">{label}</dt>
   767→      <dd className={`truncate text-foreground/85 ${mono ? "font-mono text-[10px]" : ""}`} title={value}>
   768→        {value}
   769→      </dd>
   770→    </div>
   771→  );
   772→}
