   165→   165→def reliability_curve(y, p, bins=10):
   166→   166→    out = []
   167→   167→    edges = np.linspace(0, 1, bins + 1)
   168→   168→    for i in range(bins):
   169→   169→        lo, hi = edges[i], edges[i + 1]
   170→   170→        mask = (p >= lo) & (p < hi if i < bins - 1 else p <= hi)
   171→   171→        if mask.sum() > 0:
   172→   172→            out.append({
   173→   173→                "bin_low": float(lo), "bin_high": float(hi),
   174→   174→                "mean_predicted": float(p[mask].mean()),
   175→   175→                "fraction_positive": float(y[mask].mean()),
   176→   176→                "count": int(mask.sum()),
   177→   177→            })
   178→   178→    return out

   181→   181→# ------------------------------------------------------- Platt / temp
   182→   182→def fit_platt(logits, y):
   183→   183→    """Platt scaling: p = sigmoid(a*logit + b) fitted on OOF train predictions."""
   184→   184→    from sklearn.linear_model import LogisticRegression
   185→   185→    lr = LogisticRegression(C=1e6, solver="lbfgs")
   186→   186→    lr.fit(logits.reshape(-1, 1), y)
   187→   187→    a, b = float(lr.coef_[0][0]), float(lr.intercept_[0])
   188→   188→    pc = sigmoid(a * logits + b)
   189→   189→    return a, b, float(brier_score_loss(y, pc))

   191→   191→
   192→   192→def fit_temperature(logits, y):
   193→   193→    """Temperature scaling (Guo et al. 2017) on OOF multiclass logits."""
   194→   194→    def nll(T):
   195→   195→        p = softmax(logits / T)
   196→   196→        return -np.mean(np.log(np.clip(p[np.arange(len(y)), y], 1e-12, 1)))
   197→   197→    res = minimize_scalar(nll, bounds=(0.2, 10.0), method="bounded")
   198→   198→    T = float(res.x)
   199→   199→    return T, float(nll(T)), float(nll(1.0))

   203→   203→def average_path_length_c(n: int) -> float:
   204→   204→    if n <= 1:
   205→   205→        return 0.0
   206→   206→    if n == 2:
   207→   207→        return 1.0
   208→   208→    return 2.0 * (np.log(n - 1.0) + np.euler_gamma) - 2.0 * (n - 1.0) / n

   211→   211→def serialize_iforest(iforest: IsolationForest, X_normal: np.ndarray):
   212→   212→    """Serialize sklearn IsolationForest into portable JSON + own scorer.
   213→   213→
   214→   214→    Anomaly score (original paper): s(x) = 2^(-E(h(x)) / c(n)),
   215→   215→    E(h) = mean over trees of (depth(x) + c(n_leaf - 1)).
   216→   216→    """
   217→   217→    trees = []
   218→   218→    for est in iforest.estimators_:
   219→   219→        t = est.tree_
   220→   220→        nodes = []
   221→   221→        for i in range(t.node_count):
   222→   222→            is_leaf = t.children_left[i] == -1
   223→   223→            nodes.append([
   224→   224→                int(t.feature[i]) if not is_leaf else -1,
   225→   225→                float(t.threshold[i]) if not is_leaf else 0.0,
   226→   226→                int(t.children_left[i]),
   227→   227→                int(t.children_right[i]),
   228→   228→                int(t.n_node_samples[i]),
   229→   229→            ])
   230→   230→        trees.append(nodes)
   231→   231→    n = int(iforest.max_samples_)
   232→   232→    # normalization anchors from TRAINING normal scores (deterministic mapping)
   233→   233→    raw = iforest_score(iforest, X_normal)
   234→   234→    anchors = {
   235→   235→        "p50": float(np.percentile(raw, 50)),
   236→   236→        "p90": float(np.percentile(raw, 90)),
   237→   237→        "p99": float(np.percentile(raw, 99)),
   238→   238→        "p999": float(np.percentile(raw, 99.9)),
   239→   239→    }
   240→   240→    return {
   241→   241→        "n_estimators": len(iforest.estimators_),
   242→   242→        "subsample_size": n,
   243→   243→        "c_n": average_path_length_c(n),
   244→   244→        "trees": trees,
   245→   245→        "norm_anchors": anchors,
   246→   246→        "norm_anchor_targets": {"p50": 15, "p90": 40, "p99": 70, "p999": 90},
   247→   247→        "score_formula": "s(x) = 2^(-E(h(x))/c(n)); normalized 0-100 piecewise-linear over training-normal percentile anchors",
   248→   248→    }

   250→   250→
   251→   251→def iforest_score(iforest: IsolationForest, X: np.ndarray) -> np.ndarray:
   252→   252→    """Reference implementation identical to the TS engine (paper formula)."""
   253→   253→    c_n = average_path_length_c(int(iforest.max_samples_))
   254→   254→    depths = np.zeros(len(X))
   255→   255→    for est in iforest.estimators_:
   256→   256→        t = est.tree_
   257→   257→        idx = est.apply(X)
   258→   258→        dp_sum = np.asarray(est.decision_path(X).sum(axis=1)).ravel()
   259→   259→        depths += dp_sum - 1 + np.array(
   260→   260→            [average_path_length_c(t.n_node_samples[i] - 1) for i in idx]
   261→   261→        )
   262→   262→    return 2.0 ** (-depths / (len(iforest.estimators_) * c_n))

   265→   265→def normalize_anomaly(raw: float, anchors: dict, targets: dict) -> float:
   266→   266→    """Piecewise-linear, deterministic mapping to 0..100 (same in TS)."""
   267→   267→    xs = [0.0, anchors["p50"], anchors["p90"], anchors["p99"], anchors["p999"], 1.0]
   268→   268→    ys = [0.0, targets["p50"], targets["p90"], targets["p99"], targets["p999"], 100.0]
   269→   269→    return float(np.interp(raw, xs, ys))

   272→   272→# ---------------------------------------------------------- risk score
   273→   273→def risk_score(attack_prob, anomaly_norm, category, related_alerts, cfg=RISK_CONFIG):
   274→   274→    w = cfg["weights"]
   275→   275→    sev = cfg["category_severity"].get(category, 0.3)
   276→   276→    prevalence = RARITY.get(category, 0.5)
   277→   277→    rarity = min(1.0, -np.log10(max(prevalence, 1e-6)) / 4.0)
   278→   278→    unc = 2 * attack_prob * (1 - attack_prob) * 2
   279→   279→    corr = min(1.0, related_alerts / cfg["correlation_alert_saturation"])
   280→   280→    risk = (
   281→   281→        w["attack_confidence"] * attack_prob
   282→   282→        + w["anomaly"] * (anomaly_norm / 100.0)
   283→   283→        + w["category_severity"] * sev
   284→   284→        + w["rarity"] * rarity
   285→   285→        + w["uncertainty"] * unc
   286→   286→        + w["correlation"] * corr
   287→   287→    )
   288→   288→    return float(min(100.0, max(0.0, risk)))

   302→   302→def iforest_score_serialized(trees: list, X: np.ndarray, c_n: float, n_est: int) -> np.ndarray:
   303→   303→    """Score using the SERIALIZED tree format — identical to the TS engine.
   304→   304→    Used for resume + guarantees python/TS scoring consistency."""
   305→   305→    EULER_GAMMA = 0.5772156649015329
   306→   306→
   307→   307→    def c(n):
   308→   308→        if n <= 1:
   309→   309→            return 0.0
   310→   310→        if n == 2:
   311→   311→            return 1.0
   312→   312→        return 2.0 * (np.log(n - 1) + EULER_GAMMA) - 2.0 * (n - 1) / n
   313→   313→
   314→   314→    depths = np.zeros(len(X))
   315→   315→    for nodes in trees:
   316→   316→        feats = np.array([n[0] for n in nodes])
   317→   317→        thrs = np.array([n[1] for n in nodes])
   318→   318→        lefts = np.array([n[2] for n in nodes], dtype=int)
   319→   319→        rights = np.array([n[3] for n in nodes], dtype=int)
   320→   320→        nsamp = np.array([n[4] for n in nodes], dtype=int)
   321→   321→        idx = np.zeros(len(X), dtype=int)
   322→   322→        depth = np.zeros(len(X), dtype=float)
   323→   323→        active = np.ones(len(X), dtype=bool)
   324→   324→        while active.any():
   325→   325→            f = feats[idx]
   326→   326→            is_leaf = f == -1
   327→   327→            go_left = X[np.arange(len(X)), f] <= thrs[idx]
   328→   328→            nxt = np.where(go_left, lefts[idx], rights[idx])
   329→   329→            depths[active & is_leaf] += depth[active & is_leaf] + np.array([c(nsamp[i] - 1) for i in idx[active & is_leaf]])
   330→   330→            depth = depth + (~is_leaf)
   331→   331→            idx = np.where(is_leaf, idx, nxt)
   332→   332→            active = active & ~is_leaf
   333→   333→    return 2.0 ** (-depths / (n_est * c_n))
