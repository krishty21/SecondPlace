    92→def sigmoid(x):

    96→def softmax(z):
    97→    z = z - z.max(axis=1, keepdims=True)

   118→def binary_metrics(y, p, thr=0.5, prefix=""):
   119→    pred = (p >= thr).astype(int)
   120→    fpr, tpr, thr_roc = roc_curve(y, p)
   121→    prec_arr, rec_arr, thr_pr = precision_recall_curve(y, p)
   122→    return {
   123→        "accuracy": float(accuracy_score(y, pred)),
   124→        "precision": float(precision_score(y, pred, zero_division=0)),
   125→        "recall": float(recall_score(y, pred, zero_division=0)),
   126→        "f1": float(f1_score(y, pred, zero_division=0)),
   127→        "roc_auc": float(roc_auc_score(y, p)),
   128→        "pr_auc": float(average_precision_score(y, p)),
   129→        "specificity": float(recall_score(y, 1 - pred, zero_division=0)),
   130→        "false_positive_rate": float(recall_score(1 - y, pred, zero_division=0)),
   131→        "confusion_matrix": confusion_matrix(y, pred).tolist(),
   132→        "threshold": float(thr),
   133→    }

   136→def multiclass_metrics(y, y_pred, probs, label=""):
   137→    per_class = []
   138→    for i, c in enumerate(CATS):
   139→        mask = y == i
   140→        if mask.sum() == 0:
   141→            per_class.append({"class": c, "support": 0, "precision": 0.0, "recall": 0.0, "f1": 0.0})
   142→            continue
   143→        p_i = (y_pred == i).astype(int)
   144→        t_i = mask.astype(int)
   145→        per_class.append({
   146→            "class": c, "support": int(mask.sum()),
   147→            "precision": float(precision_score(t_i, p_i, zero_division=0)),
   148→            "recall": float(recall_score(t_i, p_i, zero_division=0)),
   149→            "f1": float(f1_score(t_i, p_i, zero_division=0)),
   150→        })
   151→    cm = confusion_matrix(y, y_pred, labels=list(range(len(CATS))))
   152→    return {
   153→        "accuracy": float(accuracy_score(y, y_pred)),
   154→        "macro_precision": float(precision_score(y, y_pred, average="macro", zero_division=0)),
   155→        "macro_recall": float(recall_score(y, y_pred, average="macro", zero_division=0)),
   156→        "macro_f1": float(f1_score(y, y_pred, average="macro", zero_division=0)),
   157→        "weighted_f1": float(f1_score(y, y_pred, average="weighted", zero_division=0)),
   158→        "balanced_accuracy": float(balanced_accuracy_score(y, y_pred)),
   159→        "per_class": per_class,
   160→        "confusion_matrix": cm.tolist(),
   161→        "classes": CATS,
   162→    }

   165→def reliability_curve(y, p, bins=10):
   166→    out = []
   167→    edges = np.linspace(0, 1, bins + 1)
   168→    for i in range(bins):
   169→        lo, hi = edges[i], edges[i + 1]
   170→        mask = (p >= lo) & (p < hi if i < bins - 1 else p <= hi)
   171→        if mask.sum() > 0:
   172→            out.append({
   173→                "bin_low": float(lo), "bin_high": float(hi),
   174→                "mean_predicted": float(p[mask].mean()),
   175→                "fraction_positive": float(y[mask].mean()),
   176→                "count": int(mask.sum()),
   177→            })
   178→    return out

   181→# ------------------------------------------------------- Platt / temp
   182→def fit_platt(logits, y):
   183→    """Platt scaling: p = sigmoid(a*logit + b) fitted on OOF train predictions."""
   184→    from sklearn.linear_model import LogisticRegression
   185→    lr = LogisticRegression(C=1e6, solver="lbfgs")
   186→    lr.fit(logits.reshape(-1, 1), y)
   187→    a, b = float(lr.coef_[0][0]), float(lr.intercept_[0])
   188→    pc = sigmoid(a * logits + b)
   189→    return a, b, float(brier_score_loss(y, pc))

   192→def fit_temperature(logits, y):
   193→    """Temperature scaling (Guo et al. 2017) on OOF multiclass logits."""
   194→    def nll(T):
   195→        p = softmax(logits / T)
   196→        return -np.mean(np.log(np.clip(p[np.arange(len(y)), y], 1e-12, 1)))
   197→    res = minimize_scalar(nll, bounds=(0.2, 10.0), method="bounded")
   198→    T = float(res.x)
   199→    return T, float(nll(T)), float(nll(1.0))

   202→# ------------------------------------------------ IsolationForest export
   203→def average_path_length_c(n: int) -> float:
   204→    if n <= 1:
   205→        return 0.0
   206→    if n == 2:
   207→        return 1.0
   208→    return 2.0 * (np.log(n - 1.0) + np.euler_gamma) - 2.0 * (n - 1.0) / n

   211→def serialize_iforest(iforest: IsolationForest, X_normal: np.ndarray):
   212→    """Serialize sklearn IsolationForest into portable JSON + own scorer.
   213→
   214→    Anomaly score (original paper): s(x) = 2^(-E(h(x)) / c(n)),
   215→    E(h) = mean over trees of (depth(x) + c(n_leaf - 1)).
   216→    """
   217→    trees = []
   218→    for est in iforest.estimators_:
   219→        t = est.tree_
   220→        nodes = []
   221→        for i in range(t.node_count):
   222→            is_leaf = t.children_left[i] == -1
   223→            nodes.append([
   224→                int(t.feature[i]) if not is_leaf else -1,
   225→                float(t.threshold[i]) if not is_leaf else 0.0,
   226→                int(t.children_left[i]),
   227→                int(t.children_right[i]),
   228→                int(t.n_node_samples[i]),
   229→            ])
   230→        trees.append(nodes)
   231→    n = int(iforest.max_samples_)
   232→    # normalization anchors from TRAINING normal scores (deterministic mapping)
   233→    raw = iforest_score(iforest, X_normal)
   234→    anchors = {
   235→        "p50": float(np.percentile(raw, 50)),
   236→        "p90": float(np.percentile(raw, 90)),
   237→        "p99": float(np.percentile(raw, 99)),
   238→        "p999": float(np.percentile(raw, 99.9)),
   239→    }
   240→    return {
   241→        "n_estimators": len(iforest.estimators_),
   242→        "subsample_size": n,
   243→        "c_n": average_path_length_c(n),
   244→        "trees": trees,
   245→        "norm_anchors": anchors,
   246→        "norm_anchor_targets": {"p50": 15, "p90": 40, "p99": 70, "p999": 90},
   247→        "score_formula": "s(x) = 2^(-E(h(x))/c(n)); normalized 0-100 piecewise-linear over training-normal percentile anchors",
   248→    }

   250→
   251→def iforest_score(iforest: IsolationForest, X: np.ndarray) -> np.ndarray:
   252→    """Reference implementation identical to the TS engine (paper formula)."""
   253→    c_n = average_path_length_c(int(iforest.max_samples_))
   254→    depths = np.zeros(len(X))
   255→    for est in iforest.estimators_:
   256→        t = est.tree_
   257→        idx = est.apply(X)
   258→        dp_sum = np.asarray(est.decision_path(X).sum(axis=1)).ravel()
   259→        depths += dp_sum - 1 + np.array(
   260→            [average_path_length_c(t.n_node_samples[i] - 1) for i in idx]
   261→        )
   262→    return 2.0 ** (-depths / (len(iforest.estimators_) * c_n))

   265→def normalize_anomaly(raw: float, anchors: dict, targets: dict) -> float:
   266→    """Piecewise-linear, deterministic mapping to 0..100 (same in TS)."""
   267→    xs = [0.0, anchors["p50"], anchors["p90"], anchors["p99"], anchors["p999"], 1.0]
   268→    ys = [0.0, targets["p50"], targets["p90"], targets["p99"], targets["p999"], 100.0]
   269→    return float(np.interp(raw, xs, ys))

   272→# ---------------------------------------------------------- risk score
   273→def risk_score(attack_prob, anomaly_norm, category, related_alerts, cfg=RISK_CONFIG):
   274→    w = cfg["weights"]
   275→    sev = cfg["category_severity"].get(category, 0.3)
   276→    prevalence = RARITY.get(category, 0.5)
   277→    rarity = min(1.0, -np.log10(max(prevalence, 1e-6)) / 4.0)
   278→    unc = 2 * attack_prob * (1 - attack_prob) * 2
   279→    corr = min(1.0, related_alerts / cfg["correlation_alert_saturation"])
   280→    risk = (
   281→        w["attack_confidence"] * attack_prob
   282→        + w["anomaly"] * (anomaly_norm / 100.0)
   283→        + w["category_severity"] * sev
   284→        + w["rarity"] * rarity
   285→        + w["uncertainty"] * unc
   286→        + w["correlation"] * corr
   287→    )
   288→    return float(min(100.0, max(0.0, risk)))

   291→def severity_band(risk):
   292→    b = RISK_CONFIG["severity_bands"]
   293→    if risk <= b["low"]:
   294→        return "Low"
   295→    if risk <= b["medium"]:
   296→        return "Medium"
   297→    if risk <= b["high"]:

   302→def iforest_score_serialized(trees: list, X: np.ndarray, c_n: float, n_est: int) -> np.ndarray:
   303→    """Score using the SERIALIZED tree format — identical to the TS engine.
   304→    Used for resume + guarantees python/TS scoring consistency."""
   305→    EULER_GAMMA = 0.5772156649015329
   306→
   307→    def c(n):
   308→        if n <= 1:
   309→            return 0.0
   310→        if n == 2:
   311→            return 1.0
   312→        return 2.0 * (np.log(n - 1) + EULER_GAMMA) - 2.0 * (n - 1) / n
   313→
   314→    depths = np.zeros(len(X))
   315→    for nodes in trees:
   316→        feats = np.array([n[0] for n in nodes])
   317→        thrs = np.array([n[1] for n in nodes])
   318→        lefts = np.array([n[2] for n in nodes], dtype=int)
   319→        rights = np.array([n[3] for n in nodes], dtype=int)
   320→        nsamp = np.array([n[4] for n in nodes], dtype=int)
   321→        idx = np.zeros(len(X), dtype=int)
   322→        depth = np.zeros(len(X), dtype=float)
   323→        active = np.ones(len(X), dtype=bool)
   324→        while active.any():
   325→            f = feats[idx]
   326→            is_leaf = f == -1
   327→            go_left = X[np.arange(len(X)), f] <= thrs[idx]
   328→            nxt = np.where(go_left, lefts[idx], rights[idx])
   329→            depths[active & is_leaf] += depth[active & is_leaf] + np.array([c(nsamp[i] - 1) for i in idx[active & is_leaf]])
   330→            depth = depth + (~is_leaf)
   331→            idx = np.where(is_leaf, idx, nxt)
   332→            active = active & ~is_leaf
   333→    return 2.0 ** (-depths / (n_est * c_n))
